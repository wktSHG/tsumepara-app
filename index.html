// ========================================== // 詰パラ演習アプリ バックエンド (GAS) // カレンダー印・連続記録（ストリーク）対応版 // ========================================== function setupSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. 問題一覧シート let qSheet = ss.getSheetByName('問題一覧'); if (!qSheet) { qSheet = ss.getSheets()[0]; qSheet.setName('問題一覧'); } if (qSheet.getLastRow() === 0) { qSheet.appendRow([ '問題ID', '号数', 'コーナー', '問題番号', '作者', '手数', '問題画像ID', '解答画像ID1', '解答画像ID2', 'ステータス', '次回出題日', '連続正解数', '累計出題数', '累計正解数', '最終解答日' ]); } // 2. 解答履歴シート let hSheet = ss.getSheetByName('解答履歴'); if (!hSheet) { hSheet = ss.insertSheet('解答履歴'); hSheet.appendRow(['日時', '問題ID', 'ユーザー', '正誤', '解答時間秒']); } // 3. 設定シート let sSheet = ss.getSheetByName('設定'); if (!sSheet) { sSheet = ss.insertSheet('設定'); const ownerEmail = Session.getActiveUser().getEmail() || ''; sSheet.appendRow(['キー', '値']); sSheet.appendRow(['PIN', '1234']); sSheet.appendRow(['許可メール', ownerEmail]); sSheet.appendRow(['1日の出題数', '10']); sSheet.appendRow(['ColabURL', '']); sSheet.appendRow(['1日のスタンプ目標数', '5']); } // 4. 処理済みPDFシート let pSheet = ss.getSheetByName('処理済みPDF'); if (!pSheet) { pSheet = ss.insertSheet('処理済みPDF'); pSheet.appendRow(['ファイル名', '処理日時', '登録問題数']); } } // 設定値を取得 function getSettingsMap() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sSheet = ss.getSheetByName('設定'); const data = sSheet.getDataRange().getValues(); const map = {}; for (let i = 1; i < data.length; i++) { map[String(data[i][0]).trim()] = String(data[i][1]).trim(); } return map; } // 日時セルを日本時間の yyyy-MM-dd に正規化 function formatDateKeyJst(val) { if (!val) return ''; if (val instanceof Date) { return Utilities.formatDate(val, 'Asia/Tokyo', 'yyyy-MM-dd'); } const s = String(val).trim(); const m = s.match(/^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/); if (m) { return m[1] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[3]).slice(-2); } const d = new Date(s); if (!isNaN(d.getTime())) { return Utilities.formatDate(d, 'Asia/Tokyo', 'yyyy-MM-dd'); } return ''; } // yyyy-MM-dd に対して日数を加減算 function addDaysToKey(dateKey, deltaDays) { const parts = dateKey.split('-').map(Number); const dt = new Date(Date.UTC(parts[0], parts[1] - 1, parts[2] + deltaDays)); const y = dt.getUTCFullYear(); const m = ('0' + (dt.getUTCMonth() + 1)).slice(-2); const d = ('0' + dt.getUTCDate()).slice(-2); return y + '-' + m + '-' + d; } // 解答履歴から日別解答数・連続記録を集計 function getStreakData(goalCount) { const goal = Math.max(1, parseInt(goalCount || '5', 10) || 5); const ss = SpreadsheetApp.getActiveSpreadsheet(); const hSheet = ss.getSheetByName('解答履歴'); const dailyCounts = {}; if (hSheet && hSheet.getLastRow() > 1) { const rows = hSheet.getRange(2, 1, hSheet.getLastRow() - 1, 1).getValues(); for (let i = 0; i < rows.length; i++) { const key = formatDateKeyJst(rows[i][0]); if (!key) continue; dailyCounts[key] = (dailyCounts[key] || 0) + 1; } } const todayKey = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd'); const todayCount = dailyCounts[todayKey] || 0; const achievedDates = Object.keys(dailyCounts) .filter(k => dailyCounts[k] >= goal) .sort(); const achievedSet = new Set(achievedDates); const todayAchieved = achievedSet.has(todayKey); // 現在の連続記録（今日達成なら今日から遡る。未達でも昨日まで続いていれば継続中として表示） let currentStreak = 0; let cursor = todayAchieved ? todayKey : addDaysToKey(todayKey, -1); while (achievedSet.has(cursor)) { currentStreak++; cursor = addDaysToKey(cursor, -1); } // 最長連続記録 let bestStreak = 0; let run = 0; for (let i = 0; i < achievedDates.length; i++) { if (i > 0 && addDaysToKey(achievedDates[i - 1], 1) === achievedDates[i]) { run++; } else { run = 1; } if (run > bestStreak) bestStreak = run; } return { goal: goal, todayKey: todayKey, todayCount: todayCount, todayAchieved: todayAchieved, currentStreak: currentStreak, streakPendingToday: !todayAchieved && currentStreak > 0, bestStreak: bestStreak, totalAchievedDays: achievedDates.length, dailyCounts: dailyCounts }; } // Googleログインのトークンを検証し、許可ユーザーか確認 function verifyUser(idToken) { if (!idToken) throw new Error('ログインが必要です'); const res = UrlFetchApp.fetch('https://oauth2.googleapis.com/tokeninfo?id_token=' + encodeURIComponent(idToken), { muteHttpExceptions: true }); if (res.getResponseCode() !== 200) throw new Error('認証に失敗しました。再度ログインしてください'); const info = JSON.parse(res.getContentText()); const email = (info.email || '').toLowerCase(); const settings = getSettingsMap(); const allowed = (settings['許可メール'] || '').split(',').map(e => e.trim().toLowerCase()).filter(Boolean); if (!allowed.includes(email)) { throw new Error('このGoogleアカウント（' + email + '）は利用許可されていません'); } return email; } function doPost(e) { try { const req = JSON.parse(e.postData.contents); const email = verifyUser(req.idToken); const action = req.action; let result = {}; if (action === 'getDailyProblems') { result = getDailyProblems(req.excludeIds || [], req.limit || null); } else if (action === 'getImage') { result = getImageBase64(req.fileId); } else if (action === 'recordAnswer') { result = recordAnswer(email, req.problemId, req.isCorrect, req.elapsedSec); } else if (action === 'verifyPin') { const settings = getSettingsMap(); result = { ok: String(req.pin) === String(settings['PIN'] || '1234') }; } else if (action === 'getAdminData') { result = getAdminData(req.pin); } else if (action === 'deleteProblem') { result = deleteProblem(req.pin, req.problemId); } else if (action === 'resetAllRecords') { result = resetAllRecords(req.pin); } else if (action === 'updateSettings') { result = updateSettings(req.pin, req.newSettings); } else { throw new Error('不明なリクエストです'); } return ContentService.createTextOutput(JSON.stringify({ status: 'ok', data: result })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // 今日の問題を取得（復習ロジック＋連続記録データ） function getDailyProblems(excludeIds, customLimit) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const qSheet = ss.getSheetByName('問題一覧'); const settings = getSettingsMap(); const dailyLimit = customLimit || parseInt(settings['1日の出題数'] || '10', 10); const stampGoal = parseInt(settings['1日のスタンプ目標数'] || '5', 10) || 5; const rows = qSheet.getDataRange().getValues(); const todayStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd'); const excludeSet = new Set(excludeIds); const dueProblems = []; // 復習期限が来ている問題 const newProblems = []; // 未解答の問題 const otherProblems = []; // その他の問題 let totalActive = 0; let totalAnswered = 0; let totalCorrect = 0; for (let i = 1; i < rows.length; i++) { const r = rows[i]; const id = String(r[0]); const status = String(r[9] || '有効'); if (!id || status === '除外') continue; totalActive++; const tries = Number(r[12] || 0); const corrects = Number(r[13] || 0); totalAnswered += tries; totalCorrect += corrects; if (excludeSet.has(id)) continue; const nextDate = r[10] ? formatDateKeyJst(r[10]) : ''; const item = { id: id, issue: String(r[1] || ''), corner: String(r[2] || ''), number: String(r[3] || ''), author: String(r[4] || ''), moves: String(r[5] || ''), qImgId: String(r[6] || ''), aImgId1: String(r[7] || ''), aImgId2: String(r[8] || ''), streak: Number(r[11] || 0), tries: tries, corrects: corrects }; if (tries > 0 && nextDate && nextDate <= todayStr) { dueProblems.push({ item, nextDate, streak: item.streak }); } else if (tries === 0) { newProblems.push(item); } else { otherProblems.push(item); } } // 復習問題は連続正解数が少ない順（間違えた問題優先） dueProblems.sort((a, b) => a.streak - b.streak || a.nextDate.localeCompare(b.nextDate)); // 新規問題はシャッフル shuffleArray(newProblems); shuffleArray(otherProblems); const selected = []; for (const d of dueProblems) { if (selected.length < dailyLimit) selected.push(d.item); } for (const n of newProblems) { if (selected.length < dailyLimit) selected.push(n); } for (const o of otherProblems) { if (selected.length < dailyLimit) selected.push(o); } return { problems: selected, stats: { totalActive: totalActive, dueCount: dueProblems.length, totalAnswered: totalAnswered, accuracy: totalAnswered > 0 ? Math.round((totalCorrect / totalAnswered) * 100) : 0 }, streak: getStreakData(stampGoal), dailyLimit: parseInt(settings['1日の出題数'] || '10', 10), colabUrl: settings['ColabURL'] || '' }; } function shuffleArray(arr) { for (let i = arr.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [arr[i], arr[j]] = [arr[j], arr[i]]; } } // 非公開画像をBase64で取得 function getImageBase64(fileId) { if (!fileId) return { dataUrl: '' }; const file = DriveApp.getFileById(fileId); const blob = file.getBlob(); const base64 = Utilities.base64Encode(blob.getBytes()); return { dataUrl: 'data:' + blob.getContentType() + ';base64,' + base64 }; } // 解答を記録（忘却曲線ロジック＋達成判定） function recordAnswer(email, problemId, isCorrect, elapsedSec) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const qSheet = ss.getSheetByName('問題一覧'); const hSheet = ss.getSheetByName('解答履歴'); const settings = getSettingsMap(); const stampGoal = parseInt(settings['1日のスタンプ目標数'] || '5', 10) || 5; const now = new Date(); const todayStr = Utilities.formatDate(now, 'Asia/Tokyo', 'yyyy-MM-dd'); hSheet.appendRow([ Utilities.formatDate(now, 'Asia/Tokyo', 'yyyy-MM-dd HH:mm:ss'), problemId, email, isCorrect ? '○' : '×', Number(elapsedSec || 0) ]); const rows = qSheet.getDataRange().getValues(); const intervals = [1, 3, 7, 14, 30, 60]; // 復習間隔（日） for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(problemId)) { let streak = Number(rows[i][11] || 0); let tries = Number(rows[i][12] || 0) + 1; let corrects = Number(rows[i][13] || 0); let addDays = 1; if (isCorrect) { corrects++; addDays = intervals[Math.min(streak, intervals.length - 1)]; streak++; } else { streak = 0; addDays = 1; // 間違えたら翌日必ず再出題 } const nextDate = new Date(now.getTime() + addDays * 24 * 60 * 60 * 1000); const nextDateStr = Utilities.formatDate(nextDate, 'Asia/Tokyo', 'yyyy-MM-dd'); qSheet.getRange(i + 1, 11, 1, 5).setValues([[nextDateStr, streak, tries, corrects, todayStr]]); break; } } const streakData = getStreakData(stampGoal); const justAchievedToday = (streakData.todayCount === stampGoal); return { ok: true, streak: streakData, justAchievedToday: justAchievedToday }; } // 保護者設定用データ取得 function getAdminData(pin) { const settings = getSettingsMap(); if (String(pin) !== String(settings['PIN'] || '1234')) throw new Error('PINが正しくありません'); const ss = SpreadsheetApp.getActiveSpreadsheet(); const qSheet = ss.getSheetByName('問題一覧'); const rows = qSheet.getDataRange().getValues(); const list = []; for (let i = 1; i < rows.length; i++) { const r = rows[i]; if (!r[0] || String(r[9]) === '除外') continue; list.push({ id: String(r[0]), issue: String(r[1] || ''), corner: String(r[2] || ''), number: String(r[3] || ''), author: String(r[4] || ''), moves: String(r[5] || ''), tries: Number(r[12] || 0), corrects: Number(r[13] || 0) }); } return { problems: list, settings: settings }; } // 問題ごとの消去（除外ステータスに変更） function deleteProblem(pin, problemId) { const settings = getSettingsMap(); if (String(pin) !== String(settings['PIN'] || '1234')) throw new Error('PINが正しくありません'); const ss = SpreadsheetApp.getActiveSpreadsheet(); const qSheet = ss.getSheetByName('問題一覧'); const rows = qSheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(problemId)) { qSheet.getRange(i + 1, 10).setValue('除外'); break; } } return { ok: true }; } // 記録の一括消去 function resetAllRecords(pin) { const settings = getSettingsMap(); if (String(pin) !== String(settings['PIN'] || '1234')) throw new Error('PINが正しくありません'); const ss = SpreadsheetApp.getActiveSpreadsheet(); const hSheet = ss.getSheetByName('解答履歴'); if (hSheet.getLastRow() > 1) { hSheet.deleteRows(2, hSheet.getLastRow() - 1); } const qSheet = ss.getSheetByName('問題一覧'); if (qSheet.getLastRow() > 1) { const numRows = qSheet.getLastRow() - 1; const emptyRecords = Array.from({ length: numRows }, () => ['', 0, 0, 0, '']); qSheet.getRange(2, 11, numRows, 5).setValues(emptyRecords); } return { ok: true }; } // 設定の更新（未登録のキーは自動追記） function updateSettings(pin, newSettings) { const settings = getSettingsMap(); if (String(pin) !== String(settings['PIN'] || '1234')) throw new Error('PINが正しくありません'); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sSheet = ss.getSheetByName('設定'); const data = sSheet.getDataRange().getValues(); const updatedKeys = new Set(); for (let i = 1; i < data.length; i++) { const key = String(data[i][0]).trim(); if (newSettings[key] !== undefined) { sSheet.getRange(i + 1, 2).setValue(String(newSettings[key]).trim()); updatedKeys.add(key); } } Object.keys(newSettings).forEach(key => { if (!updatedKeys.has(key) && newSettings[key] !== undefined) { sSheet.appendRow([key, String(newSettings[key]).trim()]); } }); return { ok: true }; }
